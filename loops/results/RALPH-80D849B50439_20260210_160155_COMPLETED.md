# validar_email.py
import re

def validar_email(email: str) -> bool:
    """
    Valida email usando regex simples.
    Padrão: usuario@dominio.extensao
    """
    if not email or not isinstance(email, str):
        return False
    
    email = email.strip()
    
    # Regex simples mas efetivo
    padrao = r"^[^@\s]+@[^@\s]+\.[^@\s]+$"
    
    return bool(re.match(padrao, email))

Função implementada com regex simples `^[^@\s]+@[^@\s]+\.[^@\s]+$` que verifica:
- `^[^@\s]+` - pelo menos 1 caractere antes do @ (sem @ ou espaços)
- `@` - arroba obrigatória
- `[^@\s]+` - pelo menos 1 caractere no domínio (sem @ ou espaços)
- `\.` - ponto obrigatório
- `[^@\s]+$` - pelo menos 1 caractere na extensão (sem @ ou espaços)
